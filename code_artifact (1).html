☁️ Google Drive 自動讀寫設定 (支援 Google Sites)

✅ 專為 Google Sites 設計（免登入、無 OAuth 封鎖）：

只要將下面段短代碼貼去你 Google Drive 嘅 Apps Script 並發布成 Web App，之後每次你打開 Google Sites 都會自動讀取你 Google Drive 個試算表／Excel 檔，每次改動亦會自動即時儲存！

📋 1 分鐘設定步驟：
  1. 打開你 Google Drive，新增一個 Google 試算表 (Google Sheets)（或打開你現有嗰個 Excel/試算表）。
  2. 喺頂部選單按 擴充功能 (Extensions) > Apps Script。
  3. 清空入面原有文字，貼上右上方複製嘅代碼，按「儲存」。
  4. 按右上角藍色 部署 (Deploy) > 新增部署作業 (New deployment)。
  5. 類型揀 網頁應用程式 (Web app)，執行身分揀「我 (Me)」，誰可以存取揀「所有人 (Anyone)」，按部署並複製產生出嚟嘅 Web app URL 貼入上面輸入框即可！
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('DataStore') || ss.insertSheet('DataStore');
  var val = String(sheet.getRange('A1').getValue() || '{}');
  var callback = e && e.parameter && e.parameter.callback;
  if (callback) {
    var safeCallback = String(callback).replace(/[^a-zA-Z0-9_$]/g, '');
    return ContentService.createTextOutput(safeCallback + '(' + val + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(val).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var store = ss.getSheetByName('DataStore') || ss.insertSheet('DataStore');
  var raw = (e && e.postData && e.postData.contents) ? String(e.postData.contents) : '{}';
  store.getRange('A1').setValue(raw);

  // 同時自動更新可視化 Record 工作表，方便你隨時喺 Drive 直接打開或下載成 Excel (.xlsx)
  try {
    var payload = JSON.parse(raw);
    if (payload.tableRows && payload.tableRows.length > 0) {
      var recSheet = ss.getSheetByName('Record') || ss.insertSheet('Record');
      recSheet.clearContents();
      recSheet.getRange(1, 1, payload.tableRows.length, payload.tableRows[0].length).setValues(payload.tableRows);
    }
  } catch (err) {}

  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
